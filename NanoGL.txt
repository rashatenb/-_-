# NanoGL v5.0
# Lightweight graphics library for ESP32 (MicroPython)
#
# Features:
# - Striped buffer rendering (saves RAM, no full framebuffer needed)
# - Dirty rectangle tracking (5-10x faster partial updates)
# - LUT-cached font rendering (pre-computed lookup tables)
# - Hybrid rendering (direct TFT writes + buffered Surface drawing)
# - Memory efficient (uses memoryview and bytearray slicing)
#
# Optimized for M5StickC Plus2 (ST7789 135x240), but adaptable to other displays
#
# License: MIT
# Author: [твой GitHub никнейм]
#
# Usage:
#   from NanoGL import Surface
#   surf = Surface(tft, w=240, h=40)
#   surf.fill(0x0000)
#   surf.text(font, "Hello", 10, 10, 0xFFFF, 0x0000)
#   surf.commit(0, 0)  # Push to display at y=0
#
# Or for full-screen striped rendering:
#   def draw(surf, y_offset):
#       surf.fill(0x0000)
#       surf.text(font, "Line", 10, 10, 0xFFFF, 0x0000)
#   surf.render_full(draw, screen_h=135)


class Surface:
    """
    Optimized graphics surface for ESP32 displays.
    
    Uses striped buffer rendering to minimize RAM usage while maintaining
    fast drawing speeds. Only changed pixels are sent to display (dirty tracking).
    
    Args:
        tft: Initialized display driver (ST7789, ILI9341, etc.)
        w: Surface width in pixels (default: 240)
        h: Strip buffer height in pixels (default: 40, adjust based on RAM)
    """
    
    __slots__ = ("tft", "w", "h", "stride", "arena", "mv", "_lut",
                 "_has_set_window", "_has_write", "_has_blit",
                 "_dirty_x1", "_dirty_y1", "_dirty_x2", "_dirty_y2")
    
    def __init__(self, tft, w=240, h=40):
        self.tft = tft
        self.w = w
        self.h = h
        self.stride = w * 2
        self.arena = bytearray(w * h * 2)
        self.mv = memoryview(self.arena)
        self._lut = {}
        
        # Dirty tracking
        self._dirty_x1 = w
        self._dirty_y1 = h
        self._dirty_x2 = 0
        self._dirty_y2 = 0
        
        # Cache driver methods for faster access
        self._has_set_window = hasattr(tft, '_set_window')
        self._has_write = hasattr(tft, '_write')
        self._has_blit = hasattr(tft, 'blit_buffer')
    
    # ========== LUT (Lookup Table) for fast font rendering ==========
    
    def _lut_for(self, fg, bg):
        """Generate or retrieve cached LUT for font rendering"""
        key = (fg, bg)
        lut = self._lut.get(key)
        if lut is not None:
            return lut
        
        fg_b = bytes(((fg >> 8) & 0xFF, fg & 0xFF))
        bg_b = bytes(((bg >> 8) & 0xFF, bg & 0xFF))
        
        lut = [None] * 256
        for b in range(256):
            row = bytearray(16)
            for c in range(8):
                row[c * 2:c * 2 + 2] = fg_b if (b & (0x80 >> c)) else bg_b
            lut[b] = bytes(row)
        
        self._lut[key] = lut
        return lut
    
    def drop_luts(self):
        """Clear font LUT cache to free memory"""
        self._lut.clear()
    
    # ========== Dirty tracking ==========
    
    def _mark_dirty(self, x, y, w, h):
        """Mark region as changed"""
        if x < self._dirty_x1: self._dirty_x1 = x
        if y < self._dirty_y1: self._dirty_y1 = y
        x2 = x + w
        y2 = y + h
        if x2 > self._dirty_x2: self._dirty_x2 = x2
        if y2 > self._dirty_y2: self._dirty_y2 = y2
    
    def _reset_dirty(self):
        """Reset dirty region tracking"""
        self._dirty_x1 = self.w
        self._dirty_y1 = self.h
        self._dirty_x2 = 0
        self._dirty_y2 = 0
    
    def has_dirty(self):
        """Check if any pixels have been modified"""
        return self._dirty_x2 > self._dirty_x1 and self._dirty_y2 > self._dirty_y1
    
    # ========== Drawing primitives ==========
    
    def fill(self, color):
        """Fill entire surface with color"""
        hi = (color >> 8) & 0xFF
        lo = color & 0xFF
        self.arena[:] = bytes((hi, lo)) * (self.w * self.h)
        self._mark_dirty(0, 0, self.w, self.h)
    
    def fill_rect(self, x, y, w, h, color):
        """Draw filled rectangle"""
        if w <= 0 or h <= 0: return
        if x < 0: w += x; x = 0
        if y < 0: h += y; y = 0
        if x + w > self.w: w = self.w - x
        if y + h > self.h: h = self.h - y
        if w <= 0 or h <= 0: return
        
        hi = (color >> 8) & 0xFF
        lo = color & 0xFF
        row = bytes((hi, lo)) * w
        stride = self.stride
        buf = self.arena
        
        for yy in range(y, y + h):
            off = yy * stride + (x << 1)
            buf[off:off + (w << 1)] = row
        
        self._mark_dirty(x, y, w, h)
    
    def pixel(self, x, y, color):
        """Set a single pixel"""
        if x < 0 or y < 0 or x >= self.w or y >= self.h: return
        p = y * self.stride + (x << 1)
        self.arena[p] = (color >> 8) & 0xFF
        self.arena[p + 1] = color & 0xFF
        self._mark_dirty(x, y, 1, 1)
    
    def hline(self, x, y, w, color):
        """Draw horizontal line"""
        self.fill_rect(x, y, w, 1, color)
    
    def vline(self, x, y, h, color):
        """Draw vertical line"""
        self.fill_rect(x, y, 1, h, color)
    
    def rect(self, x, y, w, h, color):
        """Draw rectangle outline"""
        if w <= 0 or h <= 0: return
        self.hline(x, y, w, color)
        self.hline(x, y + h - 1, w, color)
        self.vline(x, y, h, color)
        self.vline(x + w - 1, y, h, color)
    
    # ========== Text rendering with LUT ==========
    
    def text(self, font, s, x, y, fg, bg):
        """
        Draw text using VGA font module.
        
        Args:
            font: Font module (e.g., vga1_16x16)
            s: String to draw
            x, y: Position
            fg: Foreground color (RGB565)
            bg: Background color (RGB565)
        """
        if not s: return
        
        lut = self._lut_for(fg, bg)
        W = font.WIDTH
        H = font.HEIGHT
        first = font.FIRST
        data = font.FONT
        bpr = W >> 3
        stride = self.stride
        buf = self.arena
        total_w = len(s) * W
        
        for i, char in enumerate(s):
            code = ord(char) if isinstance(char, str) else char
            if code < first: code = 32
            char_x = x + i * W
            if char_x < 0 or char_x + W > self.w: continue
            
            base = (code - first) * H * bpr
            x0 = char_x << 1
            
            for row in range(H):
                yy = y + row
                if yy < 0 or yy >= self.h: continue
                p = yy * stride + x0
                for k in range(bpr):
                    buf[p:p + 16] = lut[data[base + row * bpr + k]]
                    p += 16
        
        self._mark_dirty(x, y, total_w, font.HEIGHT)
    
    def text_width(self, font, s):
        """Calculate text width in pixels"""
        return len(s) * font.WIDTH
    
    # ========== Output to display ==========
    
    def commit(self, x=0, y=0):
        """
        Push entire surface buffer to display at position (x, y).
        Use this when you want to draw the full surface.
        """
        tft = self.tft
        x2 = x + self.w - 1
        y2 = y + self.h - 1
        
        if self._has_set_window and self._has_write:
            tft._set_window(x, y, x2, y2)
            tft._write(self.mv)
            return
        
        if self._has_blit:
            tft.blit_buffer(self.mv, x, y, self.w, self.h)
            return
        
        tft.set_window(x, y, x2, y2)
        tft.write(self.mv)
    
    def commit_dirty(self, x_offset=0, y_offset=0):
        """
        Push only changed pixels to display (dirty rect).
        Use this for partial updates - much faster than commit().
        
        Args:
            x_offset, y_offset: Position offset on display
        """
        if not self.has_dirty():
            return
        
        x = self._dirty_x1
        y = self._dirty_y1
        w = self._dirty_x2 - x
        h = self._dirty_y2 - y
        
        # Clipping
        if x < 0: w += x; x = 0
        if y < 0: h += y; y = 0
        if x + w > self.w: w = self.w - x
        if y + h > self.h: h = self.h - y
        
        if w <= 0 or h <= 0:
            self._reset_dirty()
            return
        
        tft = self.tft
        x2 = x + x_offset + w - 1
        y2 = y + y_offset + h - 1
        
        # If width = full arena, use memoryview slice (faster)
        if w == self.w:
            start = y * self.stride
            end = start + h * self.stride
            sub_mv = self.mv[start:end]
            
            if self._has_set_window and self._has_write:
                tft._set_window(x + x_offset, y + y_offset, x2, y2)
                tft._write(sub_mv)
            elif self._has_blit:
                tft.blit_buffer(sub_mv, x + x_offset, y + y_offset, w, h)
            else:
                tft.set_window(x + x_offset, y + y_offset, x2, y2)
                tft.write(sub_mv)
        else:
            # For narrow rectangles - temporary buffer
            row_bytes = w << 1
            tmp = bytearray(row_bytes * h)
            for yy in range(h):
                src = (y + yy) * self.stride + (x << 1)
                dst = yy * row_bytes
                tmp[dst:dst + row_bytes] = self.arena[src:src + row_bytes]
            
            tmp_mv = memoryview(tmp)
            
            if self._has_set_window and self._has_write:
                tft._set_window(x + x_offset, y + y_offset, x2, y2)
                tft._write(tmp_mv)
            elif self._has_blit:
                tft.blit_buffer(tmp_mv, x + x_offset, y + y_offset, w, h)
            else:
                tft.set_window(x + x_offset, y + y_offset, x2, y2)
                tft.write(tmp_mv)
        
        self._reset_dirty()
    
    # ========== Striped buffer rendering ==========
    
    def render_full(self, draw_func, screen_h=135):
        """
        Render full screen using horizontal strips to save RAM.
        
        Args:
            draw_func: Function(surf, y_offset) that draws content
            screen_h: Total screen height in pixels
        
        The draw_func should only draw content that falls within
        [y_offset, y_offset + surf.h) range.
        """
        for y in range(0, screen_h, self.h):
            h = min(self.h, screen_h - y)
            
            if h != self.h:
                # Temporarily change arena height for last strip
                old_h = self.h
                old_arena = self.arena
                old_mv = self.mv
                
                self.h = h
                self.arena = bytearray(self.w * h * 2)
                self.mv = memoryview(self.arena)
                self._reset_dirty()
                
                draw_func(self, y)
                if self.has_dirty():
                    self.commit(0, y)
                
                self.h = old_h
                self.arena = old_arena
                self.mv = old_mv
            else:
                self.fill(0x0000)  # Clear strip
                self._reset_dirty()
                draw_func(self, y)
                if self.has_dirty():
                    self.commit(0, y)
